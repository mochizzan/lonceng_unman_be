Fase 2 — Transformasi Koordinat

Kode: flipY = pageHeight - span.Y di pdf_reader.go

Membalik sumbu Y dari origin PDF (kiri bawah) ke origin layar (kiri atas).

// Sistem koordinat PDF memiliki origin di kiri bawah
// Sistem koordinat layar memiliki origin di kiri atas
// Transformasi: flipY = pageHeight - span.Y

func flipYCoord(span Span, pageHeight float64) float64 {
    return pageHeight - span.Y
}
PDF Y (bawah) Layar Y (atas)
Fase 3 — Pengelompokan Baris

Rumus: roundY(y) = float64(int(y*4))/4.0 di pdf_reader.go

Membulatkan Y ke 0.25 terdekat untuk mengelompokkan kata pada garis horizontal yang sama.

// Kata pada baris yang sama memiliki nilai Y yang hampir sama
// Pembulatan ke 0.25 terdekat menciptakan toleransi untuk ketidaksejajaran
// Rumus: roundY(y) = float64(int(y*4)) / 4.0

func roundY(y float64) float64 {
    return float64(int(y*4)) / 4.0
}

// Contoh: 100.12 → 100.00, 100.37 → 100.25, 100.63 → 100.50
Nilai Y Mentah Y Dibulatkan (langkah 0.25) Baris Terkelompok
Fase 4 — Konstruksi Baris

Fungsi: RowToLine() di pdf_reader.go

Mengelompokkan kata berdasarkan posisi X; menggabungkan kata pada X yang sama tanpa spasi, kata pada X berbeda dengan spasi.

// RowToLine mengelompokkan kata berdasarkan koordinat X
// Kata yang berbagi X digabungkan tanpa spasi (token yang sama)
// Kata pada X berbeda digabungkan dengan spasi (token terpisah)

func RowToLine(spans []Span) string {
    // Kelompokkan berdasarkan X, urutkan berdasarkan X, gabungkan
    groups := groupByX(spans)
    sortByX(groups)
    return joinWithSpaces(groups)
}

// Input:  [{X:10,T:"IF"}, {X:10,T:"211"}, {X:50,T:"Algo"}]
// Output: "IF211 Algo"
Word Spans Token Terelompok X Baris
Fase 5 — Deteksi Kolom

Fungsi: FindColumnPositions() di pdf_reader.go

Mengelompokkan kata header berdasarkan X, mencocokkan nama kolom yang diketahui, menghitung batas sebagai titik tengah antara posisi X yang berdekatan.

// FindColumnPositions mengidentifikasi batas kolom
// 1. Kelompokkan kata header berdasarkan koordinat X
// 2. Cocokkan dengan nama kolom diketahui (NO, KODE, MATA KULIAH, dll.)
// 3. Batas = titik tengah antara posisi X yang berdekatan

func FindColumnPositions(headerRow []Span) []Column {
    xPositions := uniqueX(headerRow)
    sort.Float64s(xPositions)
    // boundary[i] = (xPositions[i] + xPositions[i+1]) / 2
    return computeBoundaries(xPositions)
}

// Contoh: X = [50, 120, 200] → batas = [85, 160]
// Kolom 1: [0, 85), Kolom 2: [85, 160), Kolom 3: [160, ∞)

Fase 6 — Ekstraksi Kolom

Fungsi: ExtractColumnsFromRow() di pdf_reader.go

Mengambil kata yang X-nya berada dalam rentang [Start, End) untuk setiap kolom.

// ExtractColumnsFromRow menetapkan setiap kata ke kolom
// berdasarkan apakah word.X berada dalam rentang [Start, End) kolom

func ExtractColumnsFromRow(row Row, cols []Column) map[string]string {
    result := make(map[string]string)
    for _, col := range cols {
        words := filterWords(row, func(w Span) bool {
            return w.X >= col.Start && w.X < col.End
        })
        result[col.Name] = joinWords(words)
    }
    return result
}

// Baris: kata pada X=10,30,55,90,150
// Kolom: [0,40)=NO, [40,100)=KODE, [100,∞)=NAMA
// Hasil: NO={10,30}, KODE={55,90}, NAMA={150}
Baris + Batas Kata per Kolom map[col]nilai
Fase 7 — Pintar Kata Split

Fungsi: SmartWordSplit() di pdf_reader.go

Memisahkan kata yang digabungkan menggunakan deteksi CamelCase, batas digit, dan deteksi singkatan (2+ huruf kapital).

// SmartWordSplit memisahkan token yang digabungkan menjadi kata
// Aturan:
// 1. CamelCase: batas lower→upper memisahkan ("IF211Algo" → "IF211" "Algo")
// 2. Batas digit: letter→digit atau digit→letter
// 3. Singkatan: 2+ huruf kapital berurutan tetap bersama

func SmartWordSplit(token string) []string {
    var words []string
    // Pindai karakter, deteksi batas:
    // - Transisi kasus: 'a' → 'A' → pisah
    // - Transisi digit: 'A' → '2' → pisah
    // - Singkatan: "IF" tetap, "IF211" pisah di digit
    return words
}

// "IF211Pemrograman" → ["IF211", "Pemrograman"]
// "SISTEMINFORMASI" → ["SISTEM", "INFORMASI"]
Token Digabung CamelCase + Digit Kata Terpisah
Fase 8 — Parsing KRS

Fungsi: parseKRSMataKuliah() di krs_parser.go

Memindai SEMUA baris, menduplikasi dengan kunci kode:kelas, mengekstrak jadwal dari kolom JADWAL.

// parseKRSMataKuliah memproses baris dokumen KRS
// - Iterasi SEMUA baris dalam dokumen
// - Menduplikasi dengan kunci komposit: kode + ":" + kelas
// - Mengekstrak info jadwal dari kolom JADWAL

func parseKRSMataKuliah(rows []Row) []MataKuliah {
    seen := make(map[string]bool)
    var courses []MataKuliah
    for _, row := range rows {
        mk := extractMK(row)
        key := mk.Kode + ":" + mk.Kelas
        if !seen[key] {
            seen[key] = true
            mk.Jadwal = parseJadwal(row["JADWAL"])
            courses = append(courses, mk)
        }
    }
    return courses
}
Semua Baris Dedup kode:kelas Mata Kuliah KRS
Fase 9 — Parsing KHS

Fungsi: parseKHSMataKuliah() di khs_parser.go

Menangani entri multi-baris: baris nama sebelumnya, data mata kuliah, baris dosen setelahnya. Menduplikasi dengan no (nomor baris).

// parseKHSMataKuliah memproses dokumen KHS dengan entri multi-baris
// Pola: [baris NAMA] → [baris data mata kuliah] → [baris DOSEN]
// Kunci deduplikasi: no (nomor urut baris)

func parseKHSMataKuliah(rows []Row) []MKHasilStudi {
    seen := make(map[int]bool)
    var results []MKHasilStudi
    i := 0
    for i < len(rows) {
        if isNamaRow(rows[i]) {
            i++ // lewati nama
            if i < len(rows) {
                mk := parseCourseRow(rows[i])
                if !seen[mk.No] {
                    seen[mk.No] = true
                    results = append(results, mk)
                }
                i++ // lewati data
            }
            if i < len(rows) && isDosenRow(rows[i]) {
                i++ // lewati dosen
            }
        } else {
            i++
        }
    }
    return results
}
Entri Multi-Baris Dedup no Mata Kuliah KHS

Fase 11 — Output JSON

Fungsi: MarshalToJSON()

Serialisasi data mata kuliah terstruktur ke format JSON untuk respons API atau penyimpanan.

// MarshalToJSON serialisasi data terstruktur ke JSON
// Input: []MataKuliah atau []MKHasilStudi
// Output: byte JSON dengan field tag yang sesuai

func MarshalToJSON(data interface{}) ([]byte, error) {
    return json.MarshalIndent(data, "", "  ")
}
